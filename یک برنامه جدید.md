### 1️⃣ کد `app.py`

```python
"""
Actor Gallery - Flask Application
نمایش بازیگران ایرانی با Flask و Tabler CSS
"""

from flask import Flask, render_template, jsonify, abort
import json
import os

app = Flask(__name__)

# مسیر فایل JSON
DATA_FILE = os.path.join(os.path.dirname(__file__), 'BAZPRO.json')

def load_actors():
    """بارگذاری داده‌های بازیگران از فایل JSON"""
    try:
        with open(DATA_FILE, 'r', encoding='utf-8') as f:
            data = json.load(f)
            # اگر داده‌ها درون یک کلید هستند، آن را استخراج کن
            if isinstance(data, dict):
                # معمولاً داده‌ها در کلیدی مثل 'actors' یا similar هستند
                # اگر مستقیماً لیست نیست، سعی کن اولین مقدار لیست را بگیری
                for key, value in data.items():
                    if isinstance(value, list):
                        return value
                return []
            return data if isinstance(data, list) else []
    except FileNotFoundError:
        print("فایل BAZPRO.json یافت نشد!")
        return []
    except json.JSONDecodeError:
        print("خطا در خواندن فایل JSON!")
        return []

@app.route('/')
def index():
    """صفحه اصلی - نمایش لیست بازیگران"""
    actors = load_actors()
    return render_template('index.html', actors=actors)

@app.route('/api/actors')
def api_actors():
    """API برای دریافت لیست بازیگران (برای فیلتر AJAX)"""
    actors = load_actors()
    return jsonify(actors)

@app.route('/actor/<actor_name>')
def actor_detail(actor_name):
    """صفحه جزئیات بازیگر"""
    actors = load_actors()
    # جستجوی بازیگر بر اساس نام (با حذف فاصله‌های اضافی)
    actor = None
    for a in actors:
        # تطبیق نام با در نظر گرفتن حالت‌های مختلف
        if a.get('name', '').strip() == actor_name.strip() or \
           a.get('name', '').replace(' ', '') == actor_name.replace(' ', ''):
            actor = a
            break
    
    if not actor:
        abort(404)
    
    return render_template('actor.html', actor=actor)

@app.errorhandler(404)
def not_found(error):
    """صفحه خطای 404"""
    return render_template('index.html', actors=[], error="بازیگر مورد نظر یافت نشد!"), 404

if __name__ == '__main__':
    app.run(debug=True, host='0.0.0.0', port=5000)
```

            $('.actor-item').each(function() {
                const gender = $(this).data('gender');
                if (gender === filter) {
                    $(this).fadeIn(300);
                } else {
                    $(this).fadeOut(300);
                }

        const value = $(this).val().toLowerCase();

            const name = $(this).data('name').toLowerCase();
